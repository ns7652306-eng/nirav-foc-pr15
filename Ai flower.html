const express = require("express"); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); app.use(express.static("public")); // Simple Iris dataset const data = [ { x: [5.1, 3.5, 1.4, 0.2], y: "setosa" }, { x: [4.9, 3.0, 1.4, 0.2], y: "setosa" }, { x: [6.4, 3.2, 4.5, 1.5], y: "versicolor" }, { x: [6.9, 3.1, 4.9, 1.5], y: "versicolor" }, { x: [6.3, 3.3, 6.0, 2.5], y: "virginica" }, { x: [5.8, 2.7, 5.1, 1.9], y: "virginica" } ]; // Distance-based "AI" classifier function predict(features) { let best = null; let smallestDistance = Infinity; for (const flower of data) { let distance = 0; for (let i = 0; i < 4; i++) { distance += Math.pow(features[i] - flower.x[i], 2); } distance = Math.sqrt(distance); if (distance < smallestDistance) { smallestDistance = distance; best = flower.y; } } return best; } // API app.post("/predict", (req, res) => { const { sepalLength, sepalWidth, petalLength, petalWidth } = req.body; const features = [ Number(sepalLength), Number(sepalWidth), Number(petalLength), Number(petalWidth) ]; if (features.some(Number.isNaN)) { return res.status(400).json({ error: "Please provide all four flower measurements." }); } const prediction = predict(features); res.json({ flower: prediction, message: `The AI predicts this flower is ${prediction}.` }); }); app.get("/", (req, res) => { res.send(` AI Flower Classifier

🌸 AI Flower Classifier

Enter the flower measurements:

`); }); app.listen(PORT, () => { console.log(`AI Flower app running on port ${PORT}`); });
